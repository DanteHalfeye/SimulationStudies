<Aside>
Particle Life is a generative simulation in which large groups of particles move according to simple interaction rules based on their types. Each particle belongs to a species, and a matrix of attraction and repulsion values determines how strongly each species is drawn toward or pushed away from every other species. Although each particle follows only these local rules, their collective behavior produces complex, emergent patterns such as clusters, swarms, rotating structures, and constantly changing ecosystems. The system demonstrates how simple interactions at the individual level can give rise to rich, unpredictable, and lifelike dynamics without any central control.
</Aside>

Test 1

Problem: The Fire Nation dominated the environment too quickly, causing all populations to converge into a single large cluster.

Adjustment: Reduced the Fire Nation's self-attraction and increased the Earth Kingdom's and Water Tribe's repulsion from the Fire Nation.

Result: Chases became longer and more dynamic, creating greater spatial diversity across the simulation.

Test 2

Problem: The Avatar was frequently trapped by the Fire Nation.

Adjustment: Increased the Avatar's repulsion from the Fire Nation and strengthened the attraction of the Earth Kingdom, Water Tribe, and Air Nomads toward the Avatar.

Result: The Avatar moved continuously throughout the environment while the other nations formed protective groups around it.

Test 3

Problem: The Non-benders formed overly compact clusters.

Adjustment: Reduced their self-attraction while maintaining a moderate attraction toward the other populations.

Result: Their movement became more dispersed, better representing civilians adapting to survive by following different groups.

Discarded Approaches
Increasing the Avatar's attraction to the other populations too much caused large, static clusters to form.
Reducing the repulsion from the Fire Nation weakened the sense of pursuit and eliminated the central tension of the simulation.

Test 4

Configuration: Final interaction matrix.

Observation: The Fire Nation forms large, cohesive clusters that move together through the environment. Earth, Water, and Air particles create smaller allied groups, while the Non-benders remain widely dispersed and frequently move between different populations.

Result: The simulation clearly conveys the intended narrative. The Fire Nation behaves as a dominant invading force, whereas the other nations organize into defensive groups instead of forming a single large cluster.

Test 5

Observation: The asymmetrical interaction matrix creates continuous motion. Fire particles pursue other populations, while Earth, Water, and the Avatar actively avoid Fire. Air Nomads remain tightly grouped and frequently stay close to the Avatar.

Result: The system produces persistent pursuit and escape dynamics rather than reaching a static equilibrium, making the ecosystem appear alive and constantly evolving.

Test 6

Observation: Different random initial positions generate noticeably different spatial organizations while preserving the same overall behaviors. In some runs the Fire Nation occupies the center of the environment, whereas in others it splits into several independent groups.

Result: The emergent patterns vary between executions, but the narrative remains consistent: the Fire Nation acts as the aggressor, the Avatar remains under constant pressure, and the remaining nations cooperate to survive. This demonstrates that the system is robust and produces multiple valid manifestations from the same set of rules.

            this.selectedRow != -1 &&
            this.selectedCol != -1) {

            let value = this.rules.get(
                this.selectedRow,
                this.selectedCol
            );

            value -= movedY * 0.01;

            value = constrain(value, -1, 1);

            this.rules.set(
                this.selectedRow,
                this.selectedCol,
                value
            );

        }

    }

    mousePressed() {

        this.selectedCol = -1;

                    this.selectedCol = c;

class Rules {

    constructor(speciesCount) {

        this.speciesCount = speciesCount;

        this.matrix = [];

        for (let i = 0; i < speciesCount; i++) {

            this.matrix[i] = [];

            for (let j = 0; j < speciesCount; j++) {

                this.matrix[i][j] = 0;

            }

        }

        // Interaction rules
        this.set(0,0,0.5);
        this.set(1,1,0.5);
        this.set(2,2,0.5);
        this.set(3,3,0.5);
        this.set(4,4,0.5);
        this.set(5,5,0.5);

        // Population weights
        this.proportions = [
            40,
            20,
            40,
            20,
            50,
            0.1
        ];
    }

    get(from,to){
        return this.matrix[from][to];
    }

    set(from,to,value){
        this.matrix[from][to]=constrain(value,-1,1);
    }

    setProportion(species,value){
        this.proportions[species]=max(0,value);
    }

    getProportion(species){
        return this.proportions[species];
    }

    randomize(){

        for(let i=0;i<this.speciesCount;i++){

            for(let j=0;j<this.speciesCount;j++){

                this.matrix[i][j]=random(-1,1);

            }

        }

    }

}

class SpatialGrid {

  constructor(cellSize) {
    this.cellSize = cellSize;
    this.cols = ceil(1 / cellSize);
    this.rows = ceil(1 / cellSize);
    this.cells = [];
    this.clear();
  }

  clear() {
    this.cells = new Array(this.cols * this.rows);

    for (let i = 0; i < this.cells.length; i++) {
      this.cells[i] = [];
    }
  }

  index(x, y) {

    let cx = floor(x / this.cellSize);
    let cy = floor(y / this.cellSize);

    cx = (cx + this.cols) % this.cols;
    cy = (cy + this.rows) % this.rows;

    return cy * this.cols + cx;
  }

  insert(id, x, y) {
    this.cells[this.index(x, y)].push(id);
  }

  build() {

    this.clear();

    for (let i = 0; i < n; i++) {
      this.insert(i, positionsX[i], positionsY[i]);
    }
  }

  query(x, y) {

    const neighbors = [];

    let cx = floor(x / this.cellSize);
    let cy = floor(y / this.cellSize);

    for (let oy = -1; oy <= 1; oy++) {

      for (let ox = -1; ox <= 1; ox++) {

        const nx = (cx + ox + this.cols) % this.cols;
        const ny = (cy + oy + this.rows) % this.rows;

        const cell = this.cells[ny * this.cols + nx];

        neighbors.push(...cell);
      }
    }

    return neighbors;
  }

}

const positionsX = new Float32Array(n);

const velocitiesY = new Float32Array(n);

function initializeParticles() {

    let total = 0;

    for (const p of rules.proportions)
        total += p;

    for (let i = 0; i < n; i++) {

        positionsX[i] = random();
        positionsY[i] = random();

        velocitiesX[i] = random(-0.01,0.01);
        velocitiesY[i] = random(-0.01,0.01);

        const r = random(total);

        let sum = 0;

        for (let s = 0; s < m; s++) {

            sum += rules.proportions[s];

            if (r <= sum) {

                colors[i] = s;
                break;

            }

        }

    }

}

let grid;
let rules;
let matrixUI;

const palette = [];


function setup() {
  createCanvas(windowWidth, windowHeight);

  colorMode(RGB);

  rules = new Rules(m);

  matrixUI = new MatrixUI(rules, palette);

  grid = new SpatialGrid(rMax);

  for (let i = 0; i < m; i++) {
    palette.push(color(`hsl(${(360 * i) / m},100%,50%)`));
  }

  initializeParticles();
}

function draw() {
  background(0);

  updateParticles();
 

  noStroke();

  for (let i = 0; i < n; i++) {
    fill(palette[colors[i]]);
    circle(positionsX[i] * width, positionsY[i] * height, 6);
  }
   matrixUI.update();
  matrixUI.draw();
}

function updateParticles() {

  // Construir la grilla
  grid.build();

  // ======= FUERZAS =======
  for (let i = 0; i < n; i++) {

    let totalForceX = 0;
    let totalForceY = 0;

    const neighbors = grid.query(
      positionsX[i],
      positionsY[i]
    );

    for (const j of neighbors) {

      if (i === j) continue;

      let rx = positionsX[j] - positionsX[i];
      let ry = positionsY[j] - positionsY[i];

      // Mundo toroidal
      if (rx > 0.5) rx -= 1;
      if (rx < -0.5) rx += 1;

      if (ry > 0.5) ry -= 1;
      if (ry < -0.5) ry += 1;

      const r = Math.hypot(rx, ry);

      if (r === 0) continue;

      // Mantener separación mínima
      if (r < minDistance) {

        const overlap = minDistance - r;

        totalForceX -= rx / r * overlap * 20;
        totalForceY -= ry / r * overlap * 20;

        continue;
      }

      if (r < rMax) {

        const f = force(
          r / rMax,
          rules.get(colors[i], colors[j])
        );

        totalForceX += rx / r * f;
        totalForceY += ry / r * f;
      }

    }

    totalForceX *= rMax * forceFactor;
    totalForceY *= rMax * forceFactor;

    // Aplicar fricción y aceleración UNA SOLA VEZ
    velocitiesX[i] =
      velocitiesX[i] * frictionFactor +
      totalForceX * dt;

    velocitiesY[i] =
      velocitiesY[i] * frictionFactor +
      totalForceY * dt;

    // Limitar velocidad
    const speed = Math.hypot(
      velocitiesX[i],
      velocitiesY[i]
    );

    if (speed > maxSpeed) {

      velocitiesX[i] *= maxSpeed / speed;
      velocitiesY[i] *= maxSpeed / speed;

    }

  }

  // ======= POSICIONES =======

  for (let i = 0; i < n; i++) {

    positionsX[i] += velocitiesX[i] * dt;
    positionsY[i] += velocitiesY[i] * dt;

    if (positionsX[i] < 0) positionsX[i] += 1;
    if (positionsX[i] >= 1) positionsX[i] -= 1;

    if (positionsY[i] < 0) positionsY[i] += 1;
    if (positionsY[i] >= 1) positionsY[i] -= 1;

  }

  // ======= CORRECCIÓN DE COLISIONES =======

  grid.build();

  for (let i = 0; i < n; i++) {

    const neighbors = grid.query(
      positionsX[i],
      positionsY[i]
    );

    for (const j of neighbors) {

      if (j <= i) continue;

      let rx = positionsX[j] - positionsX[i];
      let ry = positionsY[j] - positionsY[i];

      if (rx > 0.5) rx -= 1;
      if (rx < -0.5) rx += 1;

      if (ry > 0.5) ry -= 1;
      if (ry < -0.5) ry += 1;

      const r = Math.hypot(rx, ry);

      if (r > 0 && r < minDistance) {

        const overlap = (minDistance - r) * 0.5;

        const nx = rx / r;
        const ny = ry / r;

        positionsX[i] -= nx * overlap;
        positionsY[i] -= ny * overlap;

        positionsX[j] += nx * overlap;
        positionsY[j] += ny * overlap;

      }

    }

  }

}

function force(r, a) {

  const beta = 0.25;

  if (r < beta) {

    return 2 * (r / beta - 1);

  }

  if (r < 1) {

    return a *
      (1 - Math.abs(
        2 * r - 1 - beta
      ) / (1 - beta));

  }

  return 0;

}
